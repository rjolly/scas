import scas.polynomial.tree.PolynomialWithGB
import scas.power.DegreeReverseLexicographic
import scas.variable.Variable
import scas.base.BigInteger
import BigInteger.given

val n = 4
val r = PolynomialWithGB(using BigInteger, DegreeReverseLexicographic(0)((for (i <- 0 until n) yield Variable("a", 0, i))*))
import r.given
val a = (for (i <- 0 until n) yield {
  (0 until (if i < n - 1 then n else 1)).foldLeft(if i == n - 1 then r.one else r.zero) { (l, j) =>
    l + (0 to i).foldLeft(r.one) { (m, k) =>
      m * r.generator((j + k) % n)
    }
  }
}).toList
var t = System.currentTimeMillis();
val b = r.gb(a*)
t = System.currentTimeMillis() - t;
println("t: " + t)
