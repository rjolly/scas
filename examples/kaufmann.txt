// See https://danielakaufmann.at/talk/2026_albatross/slides.pdf

import scala.collection.mutable.{ArrayBuffer, Queue}
import scas.residue.Residue
import scas.polynomial.TreePolynomial
import scas.variable.Variable
import scas.prettyprint.Show.given
import scas.base.BigInteger, BigInteger.self.given

val n = 4
val m = n
type Element = TreePolynomial.Element[BigInteger, Array[Int]]
val c = Array.ofDim[Element](m, m + n)
val s = Array.ofDim[Element](m, m + n)

val r = Residue.inlined(0)(BigInteger)((for i <- 0 until n yield Variable("a", 0, i))
++(for j <- 0 until m yield Variable("b", 0, j))
++((for j <- 0 until m; i <- 0 to n yield if j > 0 && i > 0 then {
  Array(Variable("c", 0, j, j + i), Variable("s", 0, j, j + i))
} else Array(Variable("s", 0, j, j + i))).flatten)*)
val q = Queue.from(r.generators)
val a = for i <- 0 until n yield q.dequeue()
val b = for j <- 0 until m yield q.dequeue()
for j <- 0 until m ; i <- 0 to n do {
  if j > 0 && i > 0 then c(j)(j + i) = q.dequeue()
  s(j)(j + i) = q.dequeue()
}
import r.given
import r.ring.pp
val eqs = ArrayBuffer.empty[Element]
r.update((for i <- 0 until n yield a(i) - a(i)\2)
++(for j <- 0 until m yield b(j) - b(j)\2)*)
for j <- 0 until m do {
  for i <- 0 to n do {
    val aji = if j > 0 && i < n then s(j - 1)(j + i) else r.zero
    val bji = if i < n then a(i) * b(j) else r.zero
    val cji = if j > 0 && i > 0 then c(j)(j + i) else r.zero
    eqs += s(j)(j + i) - (aji ^ bji ^ cji)
    if j > 0 && i < n then eqs += c(j)(j + i + 1) - (aji * bji ^ aji * cji ^ bji * cji)
  }
}
extension (x: Element) def ^ (y: Element) = x + y - 2 * x * y
val an = (0 until n).foldLeft(r.zero)((l, i) => l + a(i) * 2\i)
val bm = (0 until m).foldLeft(r.zero)((l, j) => l + b(j) * 2\j)
val sm = (0 until m).foldLeft(r.zero)((l, j) => l + s(j)(j) * 2\j)
val sn = (0 until n).foldLeft(r.zero)((l, i) => l + s(m - 1)(m + i) * 2\(m + i))
eqs += sm + sn - an * bm

var time = System.currentTimeMillis();
val g = eqs.toList.gb(true).filter(_.head._1 < s(0)(0).head._1)
time = System.currentTimeMillis() - time;
println(g.show)
println("time: " + time)
