import scala.collection.mutable.Queue
import scas.residue.BooleanAlgebra
import scas.base.Boolean.given
import scas.variable.Variable

val nlen = 2
val len = 1 << (nlen - 1)
val r = BooleanAlgebra((for i <- 0 until nlen yield Variable("n", 0, i))
++(for i <- 0 until len yield Variable("a", 1, i))
++(for i <- 0 until len yield Variable("b", 1, i))
++(for i <- 0 until len yield Variable("m", 0, i))
++(for i <- 0 until len yield Variable("a", 0, i))
++(for i <- 0 until len yield Variable("b", 0, i))
++(for i <- 0 until len yield Variable("t", 0, i))*)
val q = Queue.from(r.generators)
val n = for i <- 0 until nlen yield q.dequeue()
val a_ = for i <- 0 until len yield q.dequeue()
val b_ = for i <- 0 until len yield q.dequeue()
val m = for i <- 0 until len yield q.dequeue()
val a = for i <- 0 until len yield q.dequeue()
val b = for i <- 0 until len yield q.dequeue()
val t = for i <- 0 until len yield q.dequeue()
import r.given
val ni = (for i <- 0 until len yield (0 until nlen).foldLeft(r.one) { (nc, j) =>
  val c = 1 > ((i & (1 << j)) >> j)
  nc * (n(j) + c)
}).toList
val eqs = (for i <- 0 until len yield t(i) + ((0 until len).foldLeft(r.zero) { (s, j) =>
  if (i + j < a.length) s + a(i + j) * ni(j) else s
} + b(i)) * m(i))++(for i <- 0 until len yield a_(i) + (a(i) + (0 until len).foldLeft(r.zero) { (s, j) =>
  if (i - j >= 0) s + t(i - j) * ni(j) else s
}))++(for i <- 0 until len yield b_(i) + (b(i) + t(i))).toList
var time = System.currentTimeMillis();
val g = r.gb(false, true)(eqs*)
time = System.currentTimeMillis() - time;
println("time: " + time)
