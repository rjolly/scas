import java.util.Random
import scala.collection.mutable.{ArrayBuffer, Queue}
import scas.residue.BooleanAlgebra
import scas.prettyprint.Show.given
import scas.variable.Variable

val n = 4
val m = n
val rnd = new Random()
val p = Array.ofDim[Boolean](m + n)
def prime(n: Int) = java.math.BigInteger.probablePrime(n, rnd).intValue()
var o = prime(n) * prime(m)
for i <- 0 until m + n do {
  p(i) = (o & 1) > 0
  o >>= 1
}
println(p.toList)
type Element = BooleanAlgebra.Element[Array[Int]]
val c = Array.ofDim[Element](m, m + n)
val s = Array.ofDim[Element](m, m + n)

val r = BooleanAlgebra.binary((for i <- 0 until n yield Variable("a", 0, i))
++(for j <- 0 until m yield Variable("b", 0, j))
++(for j <- 1 until m; i <- 1 to n yield Variable("c", 0, j, j + i))
++(for j <- 0 until m; i <- 0 to n yield Variable("s", 0, j, j + i))*)
val q = Queue.from(r.generators)
val a = for i <- 0 until n yield q.dequeue()
val b = for j <- 0 until m yield q.dequeue()
for j <- 1 until m ; i <- 1 to n do c(j)(j + i) = q.dequeue()
for j <- 0 until m ; i <- 0 to n do s(j)(j + i) = q.dequeue()
import r.given
val eqs = ArrayBuffer.empty[Element]
for i <- 0 until m do eqs += s(i)(i) + r(p(i))
for i <- 0 until n do eqs += s(m - 1)(m + i) + r(p(m + i))
for j <- 0 until m do {
  for i <- 0 to n do {
    val aji = if j > 0 && i < n then s(j - 1)(j + i) else r.zero
    val bji = if i < n then a(i) * b(j) else r.zero
    val cji = if j > 0 && i > 0 then c(j)(j + i) else r.zero
    eqs += s(j)(j + i) + aji + bji + cji
    if j > 0 && i < n then eqs += c(j)(j + i + 1) + aji * bji + aji * cji + bji * cji
  }
}

var time = System.currentTimeMillis();
val g = r.gb(true)(eqs.toSeq*)
time = System.currentTimeMillis() - time;
println(g.show)
println("time: " + time)
